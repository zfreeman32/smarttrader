```python
# Add to ict_model_training.py

class WalkForwardValidator:
    """Rolling walk-forward validation for time series"""
    
    def __init__(self, train_size_months=24, test_size_months=6, step_months=3):
        self.train_size = train_size_months
        self.test_size = test_size_months
        self.step = step_months
        
    def split(self, df):
        """Generate train/test splits"""
        df = df.sort_index()
        df['month'] = df.index.to_period('M')
        months = df['month'].unique()
        
        splits = []
        for i in range(0, len(months) - self.train_size - self.test_size, self.step):
            train_end = i + self.train_size
            test_end = train_end + self.test_size
            
            train_months = months[i:train_end]
            test_months = months[train_end:test_end]
            
            train_idx = df[df['month'].isin(train_months)].index
            test_idx = df[df['month'].isin(test_months)].index
            
            splits.append((train_idx, test_idx))
            
        return splits
    
    def validate(self, model_fn, df, X_cols, y_col):
        """Run walk-forward validation"""
        results = []
        
        for fold, (train_idx, test_idx) in enumerate(self.split(df)):
            X_train = df.loc[train_idx, X_cols]
            y_train = df.loc[train_idx, y_col]
            X_test = df.loc[test_idx, X_cols]
            y_test = df.loc[test_idx, y_col]
            
            model = model_fn()
            model.fit(X_train, y_train)
            
            y_pred_proba = model.predict_proba(X_test)[:, 1]
            auc = roc_auc_score(y_test, y_pred_proba)
            
            results.append({
                'fold': fold,
                'train_start': train_idx.min(),
                'train_end': train_idx.max(),
                'test_start': test_idx.min(),
                'test_end': test_idx.max(),
                'auc': auc,
                'n_train': len(train_idx),
                'n_test': len(test_idx)
            })
            
        return pd.DataFrame(results)
```

### Feature Correlation Reduction

```python
# Add to ict_feature_engineering.py

def reduce_correlated_features(df, threshold=0.85, target_col='target'):
    """Remove highly correlated features, keeping the one most correlated with target"""
    
    # Get feature columns only
    feature_cols = [c for c in df.columns if c != target_col and df[c].dtype in ['float64', 'int64']]
    
    # Calculate correlations with target
    target_corr = df[feature_cols].corrwith(df[target_col]).abs()
    
    # Calculate feature correlations
    corr_matrix = df[feature_cols].corr().abs()
    
    # Find highly correlated pairs
    to_drop = set()
    for i in range(len(feature_cols)):
        for j in range(i+1, len(feature_cols)):
            if corr_matrix.iloc[i, j] > threshold:
                # Drop the one less correlated with target
                col_i, col_j = feature_cols[i], feature_cols[j]
                if target_corr[col_i] < target_corr[col_j]:
                    to_drop.add(col_i)
                else:
                    to_drop.add(col_j)
    
    print(f"Removing {len(to_drop)} correlated features")
    return df.drop(columns=list(to_drop))
```
